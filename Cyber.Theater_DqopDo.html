if (roll < 0.50) return { type: 'omp', amount: 50 }; if (roll < 0.75) return { type: 'key', amount: 1 }; return null; } update(dt, playerX, playerY) { if (!this.active) return; // AI: преследование игрока const dx = playerX - this.x; const dy = playerY - this.y; const dist = Math.sqrt(dx * dx + dy * dy); if (dist < 300) { this.vx = (dx / dist) * this.speed; this.vy = (dy / dist) * this.speed * 0.5; } else { this.vx *= 0.95; this.vy *= 0.95; } this.x += this.vx; this.y += this.vy; this.animationFrame += dt; this.attackCooldown = Math.max(0, this.attackCooldown - dt); } draw(ctx) { if (!this.active) return; const x = this.x; const y = this.y; const s = this.size; switch (this.type) { case 'ballerina': this.drawBallerina(ctx, x, y, s); break; case 'clown': this.drawClown(ctx, x, y, s); break; case 'barman': this.drawBarman(ctx, x, y, s); break; case 'dancer': this.drawDancer(ctx, x, y, s); break; case 'plant': this.drawPlant(ctx, x, y, s); break; case 'labyrinthMonster': this.drawLabyrinthMonster(ctx, x, y, s); break; } // HP bar ctx.fillStyle = '#00ff00'; ctx.beginPath(); ctx.arc(x - s * 0.15, y - s * 0.7, s * 0.05, 0, Math.PI * 2); ctx.fill(); ctx.beginPath(); ctx.arc(x + s * 0.15, y - s * 0.7, s * 0.05, 0, Math.PI * 2); ctx.fill(); // Туловище ctx.fillStyle = '#ffaa00'; ctx.fill(); } // Ноги ctx.fillStyle = '#ffdbac'; ctx.fillRect(x - s * 0.15, y + s * 0.4, s * 0.1, s * 0.4); ctx.fillRect(x + s * 0.05, y + s * 0.4, s * 0.1, s * 0.4); } drawBarman(ctx, x, y, s) { // Голова ctx.fillStyle = '#e8d5c4'; ctx.beginPath(); ctx.arc(x, y - s * 0.6, s * 0.3, 0, Math.PI * 2); ctx.fill(); // Шляпа ctx.fillStyle = '#000'; ctx.fillRect(x - s * 0.35, y - s * 0.9, s * 0.7, s * 0.2); // Туловище (костюм) ctx.fillStyle = '#1a1a1a'; ctx.fillRect(x - s * 0.25, y - s * 0.3, s * 0.5, s * 0.6); // Рубашка ctx.fillStyle = '#fff'; ctx.fillRect(x - s * 0.1, y - s * 0.2, s * 0.2, s * 0.4); // Ноги ctx.fillStyle = '#1a1a1a'; ctx.fillRect(x - s * 0.15, y + s * 0.3, s * 0.1, s * 0.4); ctx.fillRect(x + s * 0.05, y + s * 0.3, s * 0.1, s * 0.4); } drawDancer(ctx, x, y, s) { // Голова ctx.fillStyle = '#e8d5c4'; ctx.beginPath(); ctx.arc(x, y - s * 0.6, s * 0.3, 0, Math.PI * 2); ctx.fill(); // Туловище ctx.fillStyle = '#ff6b9d'; ctx.fillRect(x - s * 0.2, y - s * 0.3, s * 0.4, s * 0.5); // Танцевальное движение const dance = Math.sin(this.animationFrame * 0.1) * 0.2; // Левая рука ctx.strokeStyle = '#e8d5c4'; ctx.lineWidth = s * 0.08; ctx.beginPath(); ctx.moveTo(x - s * 0.2, y - s * 0.1); ctx.lineTo(x - s * 0.4 + dance, y - s * 0.3); ctx.stroke(); // Правая рука ctx.beginPath(); ctx.moveTo(x + s * 0.2, y - s * 0.1); ctx.lineTo(x + s * 0.4 - dance, y - s * 0.3); ctx.stroke(); // Ноги ctx.fillStyle = '#e8d5c4'; ctx.fillRect(x - s * 0.15, y + s * 0.2, s * 0.1, s * 0.4); ctx.fillRect(x + s * 0.05, y + s * 0.2, s * 0.1, s * 0.4); ctx.beginPath(); ctx.arc(x, y - s * 0.3, s * 0.25, 0, Math.PI * 2); ctx.fill(); // Зубы ctx.fillStyle = '#fff'; for (let i = 0; i < 6; i++) { const angle = (i / 6) * Math.PI; const tx = x + Math.cos(angle) * s * 0.3; const ty = y - s * 0.3 + Math.sin(angle) * s * 0.3; ctx.beginPath(); ctx.arc(tx, ty, s * 0.06, 0, Math.PI * 2); ctx.fill(); } } drawLabyrinthMonster(ctx, x, y, s) { // Голова (монитор) ctx.fillStyle = '#1a1a1a'; ctx.fillRect(x - s * 0.4, y - s * 0.5, s * 0.8, s * 0.5); // Экран ctx.fillStyle = '#00ff00'; ctx.globalAlpha = 0.7; ctx.fillRect(x - s * 0.35, y - s * 0.45, s * 0.7, s * 0.4); ctx.globalAlpha = 1; // Щупальца (2 штуки) ctx.strokeStyle = '#666'; ctx.lineWidth = s * 0.15; // Левое щупальце ctx.beginPath(); ctx.moveTo(x - s * 0.2, y); const leftWave = Math.sin(this.animationFrame * 0.08) * s * 0.3; ctx.quadraticCurveTo(x - s * 0.4 + leftWave, y + s * 0.3, x - s * 0.3, y + s * 0.6); ctx.stroke(); // Правое щупальце ctx.beginPath(); ctx.moveTo(x + s * 0.2, y); const rightWave = Math.sin(this.animationFrame * 0.08 + Math.PI) * s * 0.3; ctx.quadraticCurveTo(x + s * 0.4 + rightWave, y + s * 0.3, x + s * 0.3, y + s * 0.6); ctx.stroke(); // Туловище ctx.fillStyle = '#333'; ctx.beginPath(); ctx.ellipse(x, y + s * 0.3, s * 0.3, s * 0.4, 0, 0, Math.PI * 2); ctx.fill(); } takeDamage(amount) { this.hp -= amount; if (this.hp <= 0) { this.active = false; return true; } return false; } canAttack() { return this.attackCooldown <= 0; } attack() { this.attackCooldown = 1.5; return { damage: 15 + Math.random() * 10, type: this.type }; } } // ==================== БЕЛЫЙ МЕДВЕДЬ ==================== class WhiteBear { constructor(x, y) { this.x = x; this.y = y; this.size = 50; this.hp = 500; this.maxHp = 500; this.speed = 8; this.vx = 0; this.vy = 0; this.animationFrame = 0; this.attackCooldown = 0; this.active = true; this.roarPlayed = false; this.phase = 'chase'; // chase, halfAttack, killAttack } update(dt, playerX, playerY) { if (!this.active) return; const dx = playerX - this.x; const dy = playerY - this.y; const dist = Math.sqrt(dx * dx + dy * dy); // Медведь преследует игрока if (dist > 0) { this.vx = (dx / dist) * this.speed; } if (this.hp < this.maxHp * 0.2) { this.phase = 'killAttack'; } } draw(ctx) { if (!this.active) return; const x = this.x; const y = this.y; const s = this.size; // Туловище ctx.fillStyle = '#ffffff'; ctx.beginPath(); ctx.ellipse(x, y, s * 0.6, s * 0.5, 0, 0, Math.PI * 2); ctx.fill(); // Голова ctx.beginPath(); ctx.arc(x + s * 0.3, y - s * 0.3, s * 0.35, 0, Math.PI * 2); ctx.fill(); // Уши ctx.beginPath();
