def make_tone(freq=440, duration=0.12, volume=0.35, wave="square", slide_to=None):
    if not SOUND_OK:
        return None
    sr = 44100
    n = max(1, int(sr * duration))
    buf = array.array('h', [0] * (n * 2))
    amp = int(volume * 32767)
    for i in range(n):
        t = i / sr
        f = freq
        if slide_to:
            f = freq + (slide_to - freq) * (i / n)
        phase = 2 * math.pi * f * t
        if wave == "square":
            val = amp if math.sin(phase) >= 0 else -amp
        elif wave == "sine":
            val = int(amp * math.sin(phase))
        elif wave == "triangle":
            val = int(amp * (2 / math.pi) * math.asin(math.sin(phase)))
        elif wave == "noise":
            val = random.randint(-amp, amp)
        else:
            val = amp if math.sin(phase) >= 0 else -amp
        env = 1.0 - (i / n)
        val = int(val * env)
        buf[i * 2] = val
        buf[i * 2 + 1] = val
    try:
        return pygame.mixer.Sound(buffer=buf)
    except Exception:
        return None

def clamp(v, a, b):
    return max(a, min(b, v))


def shade(color, amt):
    r, g, b = color
    return (clamp(r + amt, 0, 255), clamp(g + amt, 0, 255), clamp(b + amt, 0, 255))


def lerp_color(c1, c2, t):
    return (int(c1[0] + (c2[0] - c1[0]) * t),
            int(c1[1] + (c2[1] - c1[1]) * t),
            int(c1[2] + (c2[2] - c1[2]) * t))


def draw_vgradient(surf, rect, top_c, bot_c, radius=0):
    x, y, w, h = rect
    if w <= 0 or h <= 0:
        return
    tmp = pygame.Surface((int(w), int(h)), pygame.SRCALPHA)
    steps = max(1, int(h))
    for i in range(steps):
        t = i / steps
        col = lerp_color(top_c, bot_c, t)
        pygame.draw.line(tmp, col, (0, i), (w, i))
    if radius:
        mask = pygame.Surface((int(w), int(h)), pygame.SRCALPHA)
        pygame.draw.rect(mask, (255, 255, 255, 255), (0, 0, w, h), border_radius=radius)
        tmp.blit(mask, (0, 0), special_flags=pygame.BLEND_RGBA_MIN)
    surf.blit(tmp, (x, y))


def text(surf, s, font, color, x, y, center=False, shadow=True):
    if shadow:
        sh = font.render(s, True, (0, 0, 0))
        r = sh.get_rect()
        if center:
            r.center = (x + 2, y + 2)
        else:
            r.topleft = (x + 2, y + 2)
        surf.blit(sh, r)
    img = font.render(s, True, color)
    r = img.get_rect()
    if center:
        r.center = (x, y)
    else:
        r.topleft = (x, y)
    surf.blit(img, r)
    return r

def draw_tiled_bg(surf, tile_img, offset_x, offset_y=0, alpha=255, dark=0.0):
    w, h = surf.get_size()
    tw, th = tile_img.get_size()
    ox = int(offset_x) % tw
    oy = int(offset_y) % th
    buf = pygame.Surface((w, h))
    for y in range(-th, h + th, th):
        for x in range(-tw, w + tw, tw):
            buf.blit(tile_img, (x - ox, y - oy))
    if dark > 0:
        d = pygame.Surface((w, h), pygame.SRCALPHA)
        d.fill((0, 0, 0, int(255 * dark)))
        buf.blit(d, (0, 0))
    surf.blit(buf, (0, 0))

def try_use_machine(machine, world_pos):
    if G.card_popup is not None:
        return
    if machine["cooldown"] > 0:
        show_message("РђР’РўРћРњРђРў РџР•Р Р•Р—РђР РЇР–РђР•РўРЎРЇ", 1.0)

def resolve_card_popup(dt):
    pop = G.card_popup
    if pop is None:
        return
    pop.spin_t += dt
    if pop.stage == "spin" and pop.spin_t > 0.9:
        pop.stage = "result"
        pop.success = random.random() < pop.machine["chance"]
        if pop.success:
            pop.bonus = random.uniform(pop.machine["bmin"], pop.machine["bmax"])
            tier = pop.machine["tier"]
            if tier == "basic":
                G.player.hp = min(G.player.max_hp, G.player.hp + 25)
                show_message(f"РљРђР РўРђ Р’Р«РР“Р РђРќРђ: +25 РҐРџ ({pop.bonus:.0f}%)", 1.6)
            elif tier == "advanced":
                G.player.dmg += 4
                show_message(f"РљРђР РўРђ Р’Р«РР“Р РђРќРђ: РЈР РћРќ +4 ({pop.bonus:.0f}%)", 1.6)
            else:
                G.player.max_hp += 30

                show_message(f"Р›Р•Р“Р•РќР”РђР РќРђРЇ РљРђР РўРђ: РњРђРљРЎ.РҐРџ +30 ({pop.bonus:.0f}%)", 1.6)

    if p.hop_y >= 0:
        p.hop_y = 0
        p.vy = 0
        p.on_ground = True
        p.jumps_left = 1

    if I.interact and on_island and G.card_popup is None:
        used = False
        for m in on_island.machines:
            mx, my = on_island.x + m["ox"], on_island.y + m["oy"]
            if math.hypot(p.x - mx, p.y - my) < 46 and m["cooldown"] <= 0:
                try_use_machine(m, (mx, my))
                used = True
                break

    if boss.dodge_cd > 0:

    next_x = p.ax + I.x * p.speed * 1.4
    next_y = p.ay + I.y * p.speed * 1.4
    for obs in G.obstacles:
        if abs(next_x - obs["x"]) < obs["w"] / 2 + 12 and abs(next_y - obs["y"]) < obs["h"] / 2 + 12:
            next_x, next_y = p.ax, p.ay
    p.ax, p.ay = next_x, next_y

    dist_c = math.hypot(p.ax, p.ay)
    if dist_c > ARENA_R - 40:
        ang = math.atan2(p.ay, p.ax)
        p.ax = math.cos(ang) * (ARENA_R - 40)
        p.ay = math.sin(ang) * (ARENA_R - 40)

        G.lasers.append(Laser(p.ax, p.ay, I.facing))
        G.lasers[-1].y = p.ay
        sfx.play(sfx.laser)
        p.fire_cd = 0.18

        p.fire_cd -= dt

    for l in G.lasers:
        l.x += l.dir * LASER_SPEED
        if abs(l.x - l.start_x) > LASER_RANGE * ARENA_SCALE * 0.4:
            l.dead = True
        for obs in G.obstacles:
            if abs(l.x - obs["x"]) < obs["w"] / 2 and abs(l.y - obs["y"]) < obs["h"] / 2:
                l.dead = True
        if not l.dead and boss.alive and math.hypot(l.x - boss.x, l.y - boss.y) < 70:

                m.hp -= p.dmg
                spawn_burst(m.x, m.y, (255, 150, 150))

def wrap_text(s, font, max_w):
    words = s.split(" ")
    lines = []
    cur = ""
    for w in words:
        test_line = (cur + " " + w).strip()
        if font.size(test_line)[0] > max_w and cur:
            lines.append(cur)
            cur = w
        else:
            cur = test_line
    if cur:
        lines.append(cur)
    return lines

def handle_events():
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            pygame.quit()
            sys.exit()
        elif event.type == pygame.KEYDOWN:
            if event.key == pygame.K_ESCAPE:
                if G.state in (STATE_LORE, STATE_SETTINGS):
                    G.state = STATE_MENU
            elif event.key == pygame.K_SPACE:
                I.jump_pressed = True
            elif event.key == pygame.K_e:
                I.interact = True
        elif event.type == pygame.MOUSEBUTTONDOWN and event.button == 1:
            if G.state == STATE_MENU:
                for target, rect in menu_buttons.items():
                    if rect.collidepoint(event.pos):
                        if target == STATE_PATH:
                            start_game()
                        else:
                            G.state = target
