osc.start(audioCtx.currentTime); osc.stop(audioCtx.currentTime + duration); } catch (e) { console.error('Audio error:', e); } } function playNoise(duration, volume = 0.5, filterFreq = 1000) { if (!audioCtx) initAudio();
